List<int> Lista1 = new List<int> { 6, 7, 8, 9, 10 };
List<int> Lista2 = new List<int> { 1, 2, 3, 4, 5 };

var Lista3 = Lista1.Concat(Lista2);

var crescente = Lista3.OrderBy(x => x).ToList();

var decrescente = Lista3.OrderByDescending(x => x).ToList();

Console.WriteLine("Crescente: " + string.Join(", ", crescente));
Console.WriteLine("Decrescente: " + string.Join(", ", decrescente));

Console.ReadLine();
